cmake_minimum_required(VERSION 3.25)
project(zelda_steve_bridge LANGUAGES CXX)
enable_testing()
add_library(bridge_core src/bridge/mapping.cpp src/soh_mod/client.cpp src/soh_mod/soh_api.cpp)
target_include_directories(bridge_core PUBLIC src)
if(MSVC)
  target_compile_options(bridge_core PUBLIC $<$<COMPILE_LANGUAGE:CXX>:/std:c++latest>)
  target_compile_options(bridge_core PRIVATE /W4 /WX)
else()
  target_compile_options(bridge_core PUBLIC $<$<COMPILE_LANGUAGE:CXX>:-std=c++26>)
  target_compile_options(bridge_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(bridge src/bridge/main.cpp)
target_link_libraries(bridge PRIVATE bridge_core)
add_executable(headless_mc src/headless_mc/main.cpp)
target_link_libraries(headless_mc PRIVATE bridge_core)
add_executable(loopback tests/loopback.cpp)
target_link_libraries(loopback PRIVATE bridge_core)
foreach(target bridge headless_mc loopback)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(${target} PRIVATE $<$<CONFIG:Release>:-s>)
    if(WIN32)
      target_link_options(${target} PRIVATE -static)
    endif()
  endif()
endforeach()
if(WIN32)
  target_link_libraries(loopback PRIVATE psapi)
endif()
add_test(NAME process_loopback COMMAND loopback $<TARGET_FILE:bridge> $<TARGET_FILE:headless_mc>)
set_tests_properties(process_loopback PROPERTIES TIMEOUT 15)

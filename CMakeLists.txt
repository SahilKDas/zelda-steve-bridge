cmake_minimum_required(VERSION 3.25)
project(zelda_steve_bridge LANGUAGES CXX)
enable_testing()
option(STEVE_BUILD_JNI "Build the real Minecraft shared-memory JNI adapter" ON)
add_library(bridge_core src/bridge/mapping.cpp src/soh_mod/client.cpp src/soh_mod/soh_api.cpp)
target_include_directories(bridge_core PUBLIC src)
if(MSVC)
  target_compile_options(bridge_core PUBLIC $<$<COMPILE_LANGUAGE:CXX>:/std:c++latest>)
  target_compile_options(bridge_core PRIVATE /W4 /WX)
else()
  target_compile_options(bridge_core PUBLIC $<$<COMPILE_LANGUAGE:CXX>:-std=c++26>)
  target_compile_options(bridge_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(STEVE_BUILD_JNI)
  find_package(JNI REQUIRED)
  add_library(steve_jni SHARED src/headless_mc/jni.cpp src/bridge/mapping.cpp)
  target_include_directories(steve_jni PRIVATE src ${JNI_INCLUDE_DIRS})
  if(MSVC)
    target_compile_options(steve_jni PRIVATE /std:c++latest /W4 /WX)
  else()
    target_compile_options(steve_jni PRIVATE -std=c++26 -Wall -Wextra -Wpedantic -Werror)
    target_link_options(steve_jni PRIVATE $<$<CONFIG:Release>:-s>)
    if(WIN32)
      target_link_options(steve_jni PRIVATE -static-libgcc -static-libstdc++ -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
    endif()
  endif()
endif()
add_executable(bridge src/bridge/main.cpp)
target_link_libraries(bridge PRIVATE bridge_core)
add_executable(headless_mc src/headless_mc/main.cpp)
target_link_libraries(headless_mc PRIVATE bridge_core)
add_executable(loopback tests/loopback.cpp)
target_link_libraries(loopback PRIVATE bridge_core)
add_executable(minecraft_verify tests/minecraft_verify.cpp)
target_link_libraries(minecraft_verify PRIVATE bridge_core)
if(WIN32)
  add_executable(mc_supervisor src/headless_mc/supervisor.cpp)
  target_link_libraries(mc_supervisor PRIVATE bridge_core psapi)
  if(MSVC)
    target_compile_options(mc_supervisor PRIVATE /W4 /WX)
  else()
    target_compile_options(mc_supervisor PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(mc_supervisor PRIVATE -static $<$<CONFIG:Release>:-s>)
  endif()
endif()
foreach(target bridge headless_mc loopback minecraft_verify)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(${target} PRIVATE $<$<CONFIG:Release>:-s>)
    if(WIN32)
      target_link_options(${target} PRIVATE -static)
    endif()
  endif()
endforeach()
if(WIN32)
  target_link_libraries(loopback PRIVATE psapi)
endif()
add_test(NAME process_loopback COMMAND loopback $<TARGET_FILE:bridge> $<TARGET_FILE:headless_mc>)
set_tests_properties(process_loopback PROPERTIES TIMEOUT 15)

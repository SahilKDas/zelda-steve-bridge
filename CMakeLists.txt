cmake_minimum_required(VERSION 3.25)
project(zelda_steve_bridge LANGUAGES CXX)
enable_testing()
add_library(bridge_core src/bridge/mapping.cpp src/soh_mod/client.cpp src/soh_mod/soh_api.cpp)
target_include_directories(bridge_core PUBLIC src)
if(MSVC)
  target_compile_options(bridge_core PUBLIC /std:c++latest /W4 /WX)
else()
  target_compile_options(bridge_core PUBLIC -std=c++26 -Wall -Wextra -Wpedantic -Werror)
  target_link_options(bridge_core PUBLIC $<$<CONFIG:Release>:-s>)
endif()
add_executable(bridge src/bridge/main.cpp)
target_link_libraries(bridge PRIVATE bridge_core)
add_executable(headless_mc src/headless_mc/main.cpp)
target_link_libraries(headless_mc PRIVATE bridge_core)
add_executable(loopback tests/loopback.cpp)
target_link_libraries(loopback PRIVATE bridge_core)
add_test(NAME process_loopback COMMAND loopback $<TARGET_FILE:bridge> $<TARGET_FILE:headless_mc>)
set_tests_properties(process_loopback PROPERTIES TIMEOUT 15)
